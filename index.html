async function handleAddGame(event) { event.preventDefault(); const white = document.getElementById('gameWhite').value.trim(); const black = document.getElementById('gameBlack').value.trim(); const winner = document.getElementById('gameWinner').value; const name = document.getElementById('gameName').value.trim(); const link = document.getElementById('gameLink').value.trim(); const handicap = Number(document.getElementById('gameHandicap').value) || 0; try { if (!white || !black || !winner || !name || !link) { throw new Error('Please fill in all fields'); } if (white === black) { throw new Error('White and black cannot be the same player'); } if (winner !== 'White' && winner !== 'Black') { throw new Error('Please select a winner'); } if (handicap < 0 || handicap > 9 || !Number.isInteger(handicap)) { throw new Error('Handicap must be an integer between 0 and 9'); } await eloDb.ready; const whitePlayer = await eloDb.getPlayer(white); if (!whitePlayer) { throw new Error(`White player "${white}" was not found in the player list`); } const blackPlayer = await eloDb.getPlayer(black); if (!blackPlayer) { throw new Error(`Black player "${black}" was not found in the player list`); } const winnerName = winner === 'White' ? white : black; const loserName = winner === 'White' ? black : white; await eloDb.recordMatch(winnerName, loserName, { white, black, winnerSide: winner, handicap, name, link }); allGames.unshift({ white, black, winner, handicap, name, link }); renderGames(allGames); await loadLeaderboard(); showMessage('addGameMessage', `${winnerName} beat ${loserName}. ELO updated successfully!`, true); document.getElementById('addGameForm').reset(); document.getElementById('gameHandicap').value = '0'; } catch (error) { showMessage('addGameMessage', error.message, false); } } window.addEventListener('DOMContentLoaded', async () => { bindTabs(); await importSeedDataIfNeeded(); await loadLeaderboard(); await loadGames(); const addPlayerForm = document.getElementById('addPlayerForm'); const addGameForm = document.getElementById('addGameForm'); const exportPlayersBtn = document.getElementById('exportPlayersBtn'); const exportGamesBtn = document.getElementById('exportGamesBtn'); if (addPlayerForm) { addPlayerForm.addEventListener('submit', handleAddPlayer); } if (addGameForm) { addGameForm.addEventListener('submit', handleAddGame); } if (exportPlayersBtn) { exportPlayersBtn.addEventListener('click', handleExportPlayers); } if (exportGamesBtn) { exportGamesBtn.addEventListener('click', handleExportGames); } });
